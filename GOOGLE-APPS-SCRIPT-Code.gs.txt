const RECIPIENT_EMAIL = 'HIER-PRIVATE-GMAIL-ADRESSE-EINTRAGEN';
const SENDER_NAME = 'Erinnerungsstück – Babybauch-Anfrage';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData && e.postData.contents ? e.postData.contents : '{}');

    // Einfacher Honeypot gegen Bots.
    if (data.website) {
      return ContentService.createTextOutput('OK');
    }

    if (!data.name || !data.email) {
      throw new Error('Pflichtfelder fehlen.');
    }

    const attachments = (data.files || []).slice(0, 6).map(function(file) {
      const bytes = Utilities.base64Decode(file.data || '');
      return Utilities.newBlob(bytes, file.type || 'image/jpeg', file.name || 'foto.jpg');
    });

    const subject = 'Neue Babybauch-Anfrage – ' + clean_(data.name) + (data.city ? ' aus ' + clean_(data.city) : '');
    const textBody = buildTextBody_(data);
    const htmlBody = buildHtmlBody_(data);

    GmailApp.sendEmail(
      RECIPIENT_EMAIL,
      subject,
      textBody,
      {
        htmlBody: htmlBody,
        attachments: attachments,
        replyTo: data.email,
        name: SENDER_NAME
      }
    );

    return ContentService
      .createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    console.error(err);
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function buildTextBody_(d) {
  return [
    'NEUE ANFRAGE – BABYBAUCH-ABDRUCK',
    '',
    'KONTAKT',
    'Name: ' + val_(d.name),
    'E-Mail: ' + val_(d.email),
    'Telefon / WhatsApp: ' + val_(d.phone),
    'PLZ / Ort: ' + val_(d.postal) + ' ' + val_(d.city),
    '',
    'DER ABDRUCK',
    'Bereits vorhanden: ' + val_(d.exists),
    'Maße: ' + val_(d.dimensions),
    'Zustand: ' + val_(d.condition),
    'Bilder als Anlage: ' + ((d.files || []).length),
    '',
    'GESTALTUNG',
    'Wünsche: ' + val_(d.styles),
    'Eigene Idee: ' + val_(d.idea),
    '',
    'ABWICKLUNG',
    'Transport: ' + val_(d.transport),
    'Wunschbudget: ' + val_(d.budget),
    'Weitere Hinweise: ' + val_(d.notes),
    '',
    'VERÖFFENTLICHUNG',
    val_(d.publication),
    '',
    'Eingegangen am: ' + val_(d.submittedAt),
    '',
    '--- KOMPLETTE TEXT-ZUSAMMENFASSUNG ---',
    val_(d.summary)
  ].join('\n');
}

function buildHtmlBody_(d) {
  const photoCount = (d.files || []).length;
  const rowsContact = [
    ['Name', d.name],
    ['E-Mail', d.email],
    ['Telefon / WhatsApp', d.phone],
    ['PLZ / Ort', (d.postal || '') + ' ' + (d.city || '')]
  ];
  const rowsCast = [
    ['Abdruck vorhanden', d.exists],
    ['Maße', d.dimensions],
    ['Zustand', d.condition],
    ['Bilder als Anlage', photoCount]
  ];
  const rowsDesign = [
    ['Gestaltungswünsche', d.styles],
    ['Eigene Idee', d.idea]
  ];
  const rowsProcess = [
    ['Transport', d.transport],
    ['Wunschbudget', d.budget],
    ['Weitere Hinweise', d.notes],
    ['Veröffentlichung', d.publication]
  ];

  return `
  <div style="margin:0;padding:28px;background:#f7f2eb;font-family:Arial,sans-serif;color:#332d29;">
    <div style="max-width:760px;margin:0 auto;background:#fffdf9;border:1px solid #ead9c3;border-radius:22px;overflow:hidden;">
      <div style="padding:30px 32px;background:linear-gradient(135deg,#f7ead4,#efe0c7);">
        <div style="font-family:Georgia,serif;font-size:30px;line-height:1.2;">Neue Babybauch-Anfrage 🤍</div>
        <div style="margin-top:8px;color:#6f6257;font-size:14px;">${esc_(d.submittedAt || '')}</div>
      </div>
      <div style="padding:30px 32px;">
        ${section_('Kontakt', rowsContact)}
        ${section_('Der Abdruck', rowsCast)}
        ${section_('Gestaltung', rowsDesign)}
        ${section_('Abwicklung', rowsProcess)}

        <div style="margin:26px 0;padding:18px 20px;border-radius:14px;background:#fff7ea;border:1px solid #ead4ad;">
          <b>${photoCount} Bild${photoCount === 1 ? '' : 'er'} als Anlage</b><br>
          <span style="font-size:13px;color:#6e6258;">Die Kundin/der Kunde hat die Fotos direkt im Anfrageformular hochgeladen.</span>
        </div>

        <div style="margin-top:28px;">
          <div style="font-family:Georgia,serif;font-size:20px;margin-bottom:10px;">Text-Zusammenfassung</div>
          <pre style="white-space:pre-wrap;font-family:Arial,sans-serif;font-size:13px;line-height:1.6;background:#f8f4ee;border-radius:14px;padding:18px;color:#4c443e;">${esc_(d.summary || '')}</pre>
        </div>

        <div style="margin-top:24px;font-size:12px;color:#80756c;">Antworten auf diese E-Mail gehen direkt an die im Formular angegebene E-Mail-Adresse.</div>
      </div>
    </div>
  </div>`;
}

function section_(title, rows) {
  const body = rows.map(function(row) {
    return `<tr>
      <td style="padding:8px 12px 8px 0;width:185px;color:#76695e;font-size:13px;vertical-align:top;">${esc_(row[0])}</td>
      <td style="padding:8px 0;font-size:14px;vertical-align:top;">${esc_(val_(row[1]))}</td>
    </tr>`;
  }).join('');
  return `<div style="margin-bottom:26px;">
    <div style="font-family:Georgia,serif;font-size:20px;margin-bottom:8px;color:#3c332c;">${esc_(title)}</div>
    <table role="presentation" cellspacing="0" cellpadding="0" style="width:100%;border-collapse:collapse;">${body}</table>
  </div>`;
}

function val_(value) {
  const s = String(value == null ? '' : value).trim();
  return s || '–';
}

function clean_(value) {
  return String(value == null ? '' : value).replace(/[\r\n]+/g, ' ').trim();
}

function esc_(value) {
  return String(value == null ? '' : value)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}
